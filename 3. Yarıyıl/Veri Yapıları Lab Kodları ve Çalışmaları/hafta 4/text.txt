public class LinkedList<T extends Comparable> {

    Node<T> head;

    public LinkedList() {
        head = null;
    }

    public void addToFront(T val) {
        Node<T> newNode = new Node(val);
        newNode.next = head;
        head = newNode;
    }

    public void display() {
        Node<T> iterator = head;
        while (iterator != null) {
            System.out.print(iterator + " ");
            iterator = iterator.next;
        }
        System.out.println("");
    }

    public void addToEnd(T val) {
        Node<T> newNode = new Node(val);
        Node<T> iterator = head;
        if (head == null) {
            head = newNode;
        } else {
            while (iterator.next != null) {
                iterator = iterator.next;
            }
            iterator.next = newNode;
        }
    }
    public int count(){
        Node<T> iterator=head;
        int counter=0;
        while(iterator!=null){
            iterator=iterator.next;
            counter++;
        }
        return counter;
    }
    public T findMin(){
        if(head==null)
            return null;
        Node<T> iterator=head;
        T min=iterator.value;
        while(iterator!=null){
            if(min.compareTo(iterator.value)==1)
                min=iterator.value;
            iterator=iterator.next;
        }
        return min;
        
    }
    public void inserAtGivenIndex(int index, T val){
        Node<T> newNode=new Node(val);
        if(head==null || index==0){
            newNode.next=head;
            head=newNode;
            return;
        }
        if(index>count() || index<0){
            System.out.println("index is out of bounds");
            return;
        }
        Node<T> iterator=head;
        for (int i = 0; i < index-1; i++) {
            iterator=iterator.next;
        }
        newNode.next=iterator.next;
        iterator.next=newNode;
    }

}
